rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // ===== USER DATA =====
    
    // User profiles (read by all, write by owner)
    match /users/{userId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null && request.auth.uid == userId;
    }
    
    // User collections (read/write by owner only)
    match /userCollections/{userId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }
    
    // User analytics (read/write by owner)
    match /user_analytics/{userId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }
    
    // ===== VIDEO DATA =====
    
    // Videos (read by all authenticated, write by creator)
    match /videos/{videoId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null && request.auth.uid == request.resource.data.creatorId;
      allow update, delete: if request.auth != null && request.auth.uid == resource.data.creatorId;
    }
    
    // ===== HEALTH & MONITORING =====
    
    // Health checks (read by all authenticated)
    match /health_check/{document=**} {
      allow read: if request.auth != null;
      allow write: if true; // Allow system to write health checks
    }
    
    // Doctor reports (write by all authenticated)
    match /doctor_reports/{reportId} {
      allow read, write: if request.auth != null;
    }
    
    // ===== USER INTERACTIONS =====
    
    // Notifications (read/write by owner)
    match /notifications/{userId}/items/{itemId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }
    
    // Watch history (read/write by owner)
    match /history/{userId}/items/{itemId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }
    
    // Watch later (read/write by owner)
    match /users/{userId}/watchLater/{itemId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }
    
    // ===== VS MATCHES & GAMING =====
    
    // VS Matches (read by all, write by participants)
    match /versus_matches/{matchId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null;
      allow update: if request.auth != null && (
        request.auth.uid == resource.data.challengerId ||
        request.auth.uid == resource.data.opponentId
      );
    }
    
    // VS Match Wallets (read/write by owner)
    match /vs_match_wallets/{userId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }
    
    // ===== LIVE STREAMING =====
    
    // Live streams (read by all, write by streamer)
    match /live_streams/{streamId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null && request.auth.uid == request.resource.data.streamerId;
      allow update, delete: if request.auth != null && request.auth.uid == resource.data.streamerId;
    }
    
    // Chat messages (read by all, write by authenticated)
    match /chat_messages/{messageId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null && request.auth.uid == request.resource.data.userId;
    }
    
    // ===== FEATURED CONTENT =====
    
    // Featured videos (read by all, write by admins only)
    match /featured_videos/{videoId} {
      allow read: if request.auth != null;
      allow write: if false; // Admin-only via backend
    }
    
    // ===== DEFAULT DENY =====
    
    // Deny all other access by default
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
