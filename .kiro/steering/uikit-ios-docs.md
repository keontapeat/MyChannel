## Design Patterns

### MVC (Model-View-Controller)
```swift
// Model
struct Video {
    let id: String
    let title: String
    let url: URL
}

// View
class VideoCell: UITableViewCell {
    @IBOutlet weak var titleLabel: UILabel!
    @IBOutlet weak var thumbnailImageView: UIImageView!
    
    func configure(with video: Video) {
        titleLabel.text = video.title
        // Load thumbnail
    }
}

// Controller
class VideoListViewController: UIViewController {
    @IBOutlet weak var tableView: UITableView!
    private var videos: [Video] = []
    
    override func viewDidLoad() {
        super.viewDidLoad()
        setupTableView()
        loadVideos()
    }
    
    private func setupTableView() {
        tableView.dataSource = self
        tableView.delegate = self
        tableView.register(VideoCell.self, forCellReuseIdentifier: "VideoCell")
    }
    
    private func loadVideos() {
        // Fetch videos from service
    }
}

extension VideoListViewController: UITableViewDataSource {
    func tableView(_ tableView: UITableView, numberOfRowsInSection section: Int) -> Int {
        return videos.count
    }
    
    func tableView(_ tableView: UITableView, cellForRowAt indexPath: IndexPath) -> UITableViewCell {
        let cell = tableView.dequeueReusableCell(withIdentifier: "VideoCell", for: indexPath) as! VideoCell
        cell.configure(with: videos[indexPath.row])
        return cell
    }
}
```

### MVVM (Model-View-ViewModel)
```swift
// Model
struct Video {
    let id: String
    let title: String
    let url: URL
}

// ViewModel
class VideoListViewModel {
    private(set) var videos: [Video] = []
    var onVideosUpdated: (() -> Void)?
    
    func loadVideos() {
        // Fetch videos
        VideoService.shared.fetchVideos { [weak self] result in
            switch result {
            case .success(let videos):
                self?.videos = videos
                self?.onVideosUpdated?()
            case .failure(let error):
                print("Error: \(error)")
            }
        }
    }
}

// View Controller
class VideoListViewController: UIViewController {
    @IBOutlet weak var tableView: UITableView!
    private let viewModel = VideoListViewModel()
    
    override func viewDidLoad() {
        super.viewDidLoad()
        setupBindings()
        viewModel.loadVideos()
    }
    
    private func setupBindings() {
        viewModel.onVideosUpdated = { [weak self] in
            DispatchQueue.main.async {
                self?.tableView.reloadData()
            }
        }
    }
}
```

## Auto Layout Patterns

### Programmatic Constraints
```swift
class CustomView: UIView {
    private let titleLabel = UILabel()
    private let imageView = UIImageView()
    
    override init(frame: CGRect) {
        super.init(frame: frame)
        setupViews()
    }
    
    required init?(coder: NSCoder) {
        super.init(coder: coder)
        setupViews()
    }
    
    private func setupViews() {
        addSubview(titleLabel)
        addSubview(imageView)
        
        titleLabel.translatesAutoresizingMaskIntoConstraints = false
        imageView.translatesAutoresizingMaskIntoConstraints = false
        
        NSLayoutConstraint.activate([
            imageView.topAnchor.constraint(equalTo: topAnchor, constant: 16),
            imageView.leadingAnchor.constraint(equalTo: leadingAnchor, constant: 16),
            imageView.widthAnchor.constraint(equalToConstant: 80),
            imageView.heightAnchor.constraint(equalToConstant: 80),
            
            titleLabel.topAnchor.constraint(equalTo: topAnchor, constant: 16),
            titleLabel.leadingAnchor.constraint(equalTo: imageView.trailingAnchor, constant: 16),
            titleLabel.trailingAnchor.constraint(equalTo: trailingAnchor, constant: -16)
        ])
    }
}
```

## Networking with UIKit

### URLSession Pattern
```swift
class VideoService {
    static let shared = VideoService()
    
    func fetchVideos(completion: @escaping (Result<[Video], Error>) -> Void) {
        guard let url = URL(string: "https://api.mychannel.live/videos") else {
            completion(.failure(NetworkError.invalidURL))
            return
        }
        
        URLSession.shared.dataTask(with: url) { data, response, error in
            if let error = error {
                completion(.failure(error))
                return
            }
            
            guard let data = data else {
                completion(.failure(NetworkError.noData))
                return
            }
            
            do {
                let videos = try JSONDecoder().decode([Video].self, from: data)
                completion(.success(videos))
            } catch {
                completion(.failure(error))
            }
        }.resume()
    }
}
```

### Async/Await (iOS 15+)
```swift
class VideoService {
    func fetchVideos() async throws -> [Video] {
        guard let url = URL(string: "https://api.mychannel.live/videos") else {
            throw NetworkError.invalidURL
        }
        
        let (data, _) = try await URLSession.shared.data(from: url)
        let videos = try JSONDecoder().decode([Video].self, from: data)
        return videos
    }
}

// Usage in ViewController
Task {
    do {
        let videos = await VideoService.shared.fetchVideos()
        self.videos = videos
        self.tableView.reloadData()
    } catch {
        print("Error: \(error)")
    }
}
```

## Memory Management

### Retain Cycles & Weak References
```swift
class VideoPlayerViewController: UIViewController {
    private var player: AVPlayer?
    
    override func viewDidLoad() {
        super.viewDidLoad()
        
        // ❌ BAD: Creates retain cycle
        player?.addPeriodicTimeObserver(forInterval: CMTime(seconds: 1, preferredTimescale: 1), queue: .main) { time in
            self.updateProgress(time: time)
        }
        
        // ✅ GOOD: Use weak self
        player?.addPeriodicTimeObserver(forInterval: CMTime(seconds: 1, preferredTimescale: 1), queue: .main) { [weak self] time in
            self?.updateProgress(time: time)
        }
    }
}
```

### Image Loading & Caching
```swift
class ImageCache {
    static let shared = ImageCache()
    private let cache = NSCache<NSString, UIImage>()
    
    func loadImage(from url: URL, completion: @escaping (UIImage?) -> Void) {
        let key = url.absoluteString as NSString
        
        // Check cache
        if let cachedImage = cache.object(forKey: key) {
            completion(cachedImage)
            return
        }
        
        // Download
        URLSession.shared.dataTask(with: url) { [weak self] data, _, _ in
            guard let data = data, let image = UIImage(data: data) else {
                completion(nil)
                return
            }
            
            self?.cache.setObject(image, forKey: key)
            DispatchQueue.main.async {
                completion(image)
            }
        }.resume()
    }
}
```

## Common UIKit Patterns for MyChannel

### Video Player Integration
```swift
class VideoPlayerViewController: UIViewController {
    private var player: AVPlayer?
    private var playerLayer: AVPlayerLayer?
    
    override func viewDidLoad() {
        super.viewDidLoad()
        setupPlayer()
    }
    
    private func setupPlayer() {
        guard let url = URL(string: videoURL) else { return }
        
        player = AVPlayer(url: url)
        playerLayer = AVPlayerLayer(player: player)
        playerLayer?.frame = view.bounds
        playerLayer?.videoGravity = .resizeAspect
        
        if let playerLayer = playerLayer {
            view.layer.addSublayer(playerLayer)
        }
        
        player?.play()
    }
    
    override func viewDidLayoutSubviews() {
        super.viewDidLayoutSubviews()
        playerLayer?.frame = view.bounds
    }
}
```

### Pull-to-Refresh
```swift
class VideoFeedViewController: UIViewController {
    @IBOutlet weak var tableView: UITableView!
    private let refreshControl = UIRefreshControl()
    
    override func viewDidLoad() {
        super.viewDidLoad()
        setupRefreshControl()
    }
    
    private func setupRefreshControl() {
        refreshControl.addTarget(self, action: #selector(refreshData), for: .valueChanged)
        tableView.refreshControl = refreshControl
    }
    
    @objc private func refreshData() {
        loadVideos { [weak self] in
            self?.refreshControl.endRefreshing()
        }
    }
}
```

## UIKit Best Practices

### Thread Safety
- Always update UI on the main thread
- Use `DispatchQueue.main.async` for UI updates from background threads
- Use `@MainActor` for async functions that update UI (iOS 15+)

### View Controller Lifecycle
- Setup in `viewDidLoad` (called once)
- Update data in `viewWillAppear` (called every time)
- Start animations in `viewDidAppear`
- Pause/stop in `viewWillDisappear`
- Clean up in `deinit`

### Auto Layout
- Always set `translatesAutoresizingMaskIntoConstraints = false`
- Use layout anchors over NSLayoutConstraint
- Activate constraints in batch with `NSLayoutConstraint.activate()`
- Use UIStackView for simple layouts

### Memory Management
- Use `weak` for delegates
- Use `[weak self]` in closures
- Remove observers in `deinit`
- Cancel network requests when view disappears
