## Android Architecture Patterns

### MVVM (Model-View-ViewModel)
```kotlin
// Model
data class Video(
    val id: String,
    val title: String,
    val url: String,
    val thumbnailUrl: String
)

// Repository
class VideoRepository @Inject constructor(
    private val firestore: FirebaseFirestore
) {
    suspend fun getVideos(): Result<List<Video>> = withContext(Dispatchers.IO) {
        try {
            val snapshot = firestore.collection("videos").get().await()
            val videos = snapshot.documents.mapNotNull { it.toObject(Video::class.java) }
            Result.success(videos)
        } catch (e: Exception) {
            Result.failure(e)
        }
    }
}

// ViewModel
@HiltViewModel
class VideoListViewModel @Inject constructor(
    private val repository: VideoRepository
) : ViewModel() {
    
    private val _videos = MutableStateFlow<List<Video>>(emptyList())
    val videos: StateFlow<List<Video>> = _videos.asStateFlow()
    
    private val _isLoading = MutableStateFlow(false)
    val isLoading: StateFlow<Boolean> = _isLoading.asStateFlow()
    
    init {
        loadVideos()
    }
    
    fun loadVideos() {
        viewModelScope.launch {
            _isLoading.value = true
            repository.getVideos()
                .onSuccess { _videos.value = it }
                .onFailure { /* Handle error */ }
            _isLoading.value = false
        }
    }
}

// Composable View
@Composable
fun VideoListScreen(
    viewModel: VideoListViewModel = hiltViewModel()
) {
    val videos by viewModel.videos.collectAsState()
    val isLoading by viewModel.isLoading.collectAsState()
    
    if (isLoading) {
        CircularProgressIndicator()
    } else {
        LazyColumn {
            items(videos) { video ->
                VideoItem(video = video)
            }
        }
    }
}
```

### MVI (Model-View-Intent)
```kotlin
// State
data class VideoListState(
    val videos: List<Video> = emptyList(),
    val isLoading: Boolean = false,
    val error: String? = null
)

// Intent
sealed class VideoListIntent {
    object LoadVideos : VideoListIntent()
    data class SelectVideo(val video: Video) : VideoListIntent()
}

// ViewModel
@HiltViewModel
class VideoListViewModel @Inject constructor(
    private val repository: VideoRepository
) : ViewModel() {
    
    private val _state = MutableStateFlow(VideoListState())
    val state: StateFlow<VideoListState> = _state.asStateFlow()
    
    fun processIntent(intent: VideoListIntent) {
        when (intent) {
            is VideoListIntent.LoadVideos -> loadVideos()
            is VideoListIntent.SelectVideo -> selectVideo(intent.video)
        }
    }
    
    private fun loadVideos() {
        viewModelScope.launch {
            _state.value = _state.value.copy(isLoading = true)
            repository.getVideos()
                .onSuccess { videos ->
                    _state.value = _state.value.copy(
                        videos = videos,
                        isLoading = false
                    )
                }
                .onFailure { error ->
                    _state.value = _state.value.copy(
                        error = error.message,
                        isLoading = false
                    )
                }
        }
    }
}
```

### Repository Pattern
```kotlin
interface VideoRepository {
    suspend fun getVideos(): Result<List<Video>>
    suspend fun getVideo(id: String): Result<Video>
    fun observeVideos(): Flow<List<Video>>
}

class VideoRepositoryImpl @Inject constructor(
    private val api: VideoApi,
    private val dao: VideoDao,
    private val firestore: FirebaseFirestore
) : VideoRepository {
    
    override suspend fun getVideos(): Result<List<Video>> = withContext(Dispatchers.IO) {
        try {
            // Try network first
            val videos = api.getVideos()
            dao.insertAll(videos) // Cache locally
            Result.success(videos)
        } catch (e: Exception) {
            // Fallback to cache
            val cachedVideos = dao.getAll()
            if (cachedVideos.isNotEmpty()) {
                Result.success(cachedVideos)
            } else {
                Result.failure(e)
            }
        }
    }
    
    override fun observeVideos(): Flow<List<Video>> {
        return dao.observeAll()
    }
}
```

## Room Database Pattern

```kotlin
// Entity
@Entity(tableName = "videos")
data class VideoEntity(
    @PrimaryKey val id: String,
    val title: String,
    val url: String,
    val thumbnailUrl: String,
    val createdAt: Long
)

// DAO
@Dao
interface VideoDao {
    @Query("SELECT * FROM videos ORDER BY createdAt DESC")
    fun observeAll(): Flow<List<VideoEntity>>
    
    @Query("SELECT * FROM videos WHERE id = :id")
    suspend fun getById(id: String): VideoEntity?
    
    @Insert(onConflict = OnConflictStrategy.REPLACE)
    suspend fun insertAll(videos: List<VideoEntity>)
    
    @Delete
    suspend fun delete(video: VideoEntity)
}

// Database
@Database(entities = [VideoEntity::class], version = 1)
abstract class AppDatabase : RoomDatabase() {
    abstract fun videoDao(): VideoDao
}

// Module
@Module
@InstallIn(SingletonComponent::class)
object DatabaseModule {
    
    @Provides
    @Singleton
    fun provideDatabase(@ApplicationContext context: Context): AppDatabase {
        return Room.databaseBuilder(
            context,
            AppDatabase::class.java,
            "mychannel.db"
        ).build()
    }
    
    @Provides
    fun provideVideoDao(database: AppDatabase): VideoDao {
        return database.videoDao()
    }
}
```

## Firebase Integration

```kotlin
// Authentication
class AuthRepository @Inject constructor(
    private val auth: FirebaseAuth
) {
    suspend fun signIn(email: String, password: String): Result<FirebaseUser> {
        return try {
            val result = auth.signInWithEmailAndPassword(email, password).await()
            Result.success(result.user!!)
        } catch (e: Exception) {
            Result.failure(e)
        }
    }
    
    fun getCurrentUser(): FirebaseUser? = auth.currentUser
    
    fun observeAuthState(): Flow<FirebaseUser?> = callbackFlow {
        val listener = FirebaseAuth.AuthStateListener { auth ->
            trySend(auth.currentUser)
        }
        auth.addAuthStateListener(listener)
        awaitClose { auth.removeAuthStateListener(listener) }
    }
}

// Firestore
class VideoFirestoreRepository @Inject constructor(
    private val firestore: FirebaseFirestore
) {
    fun observeVideos(): Flow<List<Video>> = callbackFlow {
        val listener = firestore.collection("videos")
            .addSnapshotListener { snapshot, error ->
                if (error != null) {
                    close(error)
                    return@addSnapshotListener
                }
                
                val videos = snapshot?.documents?.mapNotNull {
                    it.toObject(Video::class.java)
                } ?: emptyList()
                
                trySend(videos)
            }
        
        awaitClose { listener.remove() }
    }
}
```

## Android Best Practices

### Compose Best Practices
- Use `remember` for state that survives recomposition
- Use `LaunchedEffect` for suspend functions
- Use `derivedStateOf` for computed state
- Hoist state when possible
- Use `key` in LazyColumn for performance
- Avoid unnecessary recompositions

### Kotlin Best Practices
- Use data classes for models
- Use sealed classes for states
- Prefer `val` over `var`
- Use extension functions
- Use coroutines for async operations
- Use Flow for reactive streams
- Handle null safety properly

### Architecture Best Practices
- Follow MVVM or MVI pattern
- Use repository pattern for data access
- Separate UI, domain, and data layers
- Use dependency injection (Hilt)
- Write testable code
- Handle errors gracefully
- Implement offline-first when possible
