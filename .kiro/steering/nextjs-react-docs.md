#### Custom Hook Pattern
```typescript
// lib/hooks/useAuth.ts
'use client';

import { useState, useEffect } from 'react';
import { onAuthStateChanged } from 'firebase/auth';
import { auth } from '@/lib/firebase/config';

export function useAuth() {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);
  
  useEffect(() => {
    const unsubscribe = onAuthStateChanged(auth, (user) => {
      setUser(user);
      setLoading(false);
    });
    
    return unsubscribe;
  }, []);
  
  return { user, loading };
}
```

#### Service Layer Pattern
```typescript
// services/video.service.ts
import { db } from '@/lib/firebase/config';
import { collection, doc, getDoc, query, where } from 'firebase/firestore';

export async function getVideo(videoId: string): Promise<Video> {
  const docRef = doc(db, 'videos', videoId);
  const docSnap = await getDoc(docRef);
  
  if (!docSnap.exists()) {
    throw new Error('Video not found');
  }
  
  return {
    id: docSnap.id,
    ...docSnap.data()
  } as Video;
}

export async function getUserVideos(userId: string): Promise<Video[]> {
  const q = query(
    collection(db, 'videos'),
    where('userId', '==', userId)
  );
  
  const snapshot = await getDocs(q);
  return snapshot.docs.map(doc => ({
    id: doc.id,
    ...doc.data()
  })) as Video[];
}
```

## Accessibility Best Practices
- Use semantic HTML (`<nav>`, `<main>`, `<article>`)
- Add ARIA labels where needed
- Ensure keyboard navigation works
- Maintain proper heading hierarchy (h1 → h2 → h3)
- Provide alt text for images
- Use sufficient color contrast (WCAG AA minimum)
- Test with screen readers
- Support reduced motion preferences

## Security Best Practices
- Sanitize user input (prevent XSS)
- Use Content Security Policy (CSP)
- Implement CSRF protection
- Use HTTPS only
- Validate on server-side (never trust client)
- Store secrets in environment variables
- Use secure authentication (Firebase Auth)
- Implement rate limiting
- Keep dependencies updated

## MyChannel-Specific Conventions
- Use **Server Components** by default (better performance)
- Use **Client Components** only when needed (interactivity, hooks, browser APIs)
- Fetch data in Server Components (no loading states needed)
- Use **TypeScript strict mode** (`strict: true`)
- Follow **Tailwind** for styling (no CSS modules)
- Use **shadcn/ui** for base components
- Implement **responsive design** (mobile-first)
- Use **next/image** for all images
- Use **next/font** for fonts (already configured)
- Deploy as **static export** to Firebase Hosting
